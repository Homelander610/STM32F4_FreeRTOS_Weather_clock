// --- app/ui.h ---
// ==========================================
#ifndef __UI_H__
#define __UI_H__

#include <stdint.h>
#include "font.h"
#include "image.h"

#define UI_WIDTH 240
#define UI_HEIGHT 320

/*红色占8位只需要5位所以前3位是0xF8，绿色占8位只需要6位所以前2位是0xFC，蓝色占8位只需要5位所以直接取前5位，直接舍弃后3位，然后组合成16位，前11位是红色，中6位是绿色，后5位是蓝色*/
#define mkcolor(r,g,b) (((r & 0xF8) << 8) | ((g & 0xFC) << 3) | (b >>3))	//把8位RGB颜色值转换为16位RGB565格式


void ui_init(void);
void ui_fill_color(uint16_t x, uint16_t y, uint16_t width, uint16_t height, uint16_t color);
